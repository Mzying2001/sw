246 virtual void OnScroll(ScrollOrientation scrollbar, ScrollEvent event, double pos)
247 {
248 ScrollingEventArgs args(scrollbar, event, pos);
249 this->RaiseRoutedEvent(args);
250
251 if (args.cancel) {
252 return;
253 }
254
255 if (scrollbar == ScrollOrientation::Horizontal) {
256 // 水平滚动条
257 switch (event) {
258 case ScrollEvent::ThubmTrack: {
259 HorizontalScrollPos = pos;
260 break;
261 }
262 case ScrollEvent::Left: {
263 ScrollToLeft();
264 break;
265 }
266 case ScrollEvent::Right: {
267 ScrollToRight();
268 break;
269 }
270 case ScrollEvent::PageLeft: {
271 ScrollHorizontal(-GetHorizontalScrollPageSize());
272 break;
273 }
274 case ScrollEvent::PageRight: {
275 ScrollHorizontal(GetHorizontalScrollPageSize());
276 break;
277 }
278 case ScrollEvent::LineLeft: {
279 ScrollHorizontal(-_LayerScrollBarLineInterval);
280 break;
281 }
282 case ScrollEvent::LineRight: {
283 ScrollHorizontal(_LayerScrollBarLineInterval);
284 break;
285 }
286 default: {
287 break;
288 }
289 }
290 } else {
291 // 垂直滚动条
292 switch (event) {
293 case ScrollEvent::ThubmTrack: {
294 VerticalScrollPos = pos;
295 break;
296 }
297 case ScrollEvent::Bottom: {
298 ScrollToBottom();
299 break;
300 }
301 case ScrollEvent::Top: {
302 ScrollToTop();
303 break;
304 }
305 case ScrollEvent::PageUp: {
306 ScrollVertical(-GetVerticalScrollPageSize());
307 break;
308 }
309 case ScrollEvent::PageDown: {
310 ScrollVertical(GetVerticalScrollPageSize());
311 break;
312 }
313 case ScrollEvent::LineUp: {

315 break;

407 virtual bool RequestBringIntoView(const sw::Rect &screenRect) override
408 {
409 bool handled = false;
410
411 sw::Point pos = this->PointFromScreen(screenRect.GetPos());
412 sw::Rect rect = {pos.x, pos.y, screenRect.width, screenRect.height};
413
414 sw::Rect clientRect = this->ClientRect;
415
416 if (VerticalScrollBar) //
417 {
418 double curPos = VerticalScrollPos;
419
420 if (rect.top < curPos) {
421 VerticalScrollPos = rect.top;
422 } else if (rect.top + rect.height > curPos + clientRect.height) {
423 if (rect.height >= clientRect.height) {
424 VerticalScrollPos = rect.top;
425 } else {
426 VerticalScrollPos = rect.top + rect.height - clientRect.height;
427 }
428 }
429 handled = true;
430 }
431
432 if (HorizontalScrollBar) //
433 {
434 double curPos = HorizontalScrollPos;
435
436 if (rect.left < curPos) {
437 HorizontalScrollPos = rect.left;
438 } else if (rect.left + rect.width > curPos + clientRect.width) {
439 if (rect.width >= clientRect.width) {
440 HorizontalScrollPos = rect.left;
441 } else {
442 HorizontalScrollPos = rect.left + rect.width - clientRect.width;
443 }
444 }
445 handled = true;
446 }
447
448 return handled;
449 }

594 void UpdateScrollRange()
595 {
596 if (_GetLayout() == nullptr) {
597 // 当未设置布局方式时滚动条和控件位置需要手动设置
598 // 将以下俩字段设为false确保xxxScrollLimit属性在未设置布局方式时仍可用
599 _horizontalScrollDisabled = false;
600 _verticalScrollDisabled = false;
601 return;
602 }
603
604 if (HorizontalScrollBar) //
605 {
606 double childRightmost = this->GetChildRightmost(true);
607
608 if (int(childRightmost - this->ClientWidth) > 0) {
609 _horizontalScrollDisabled = false;
610 EnableScrollBar(this->Handle, SB_HORZ, ESB_ENABLE_BOTH);
611 SetHorizontalScrollRange(0, childRightmost);
612
613 // 当尺寸改变时确保子元素位置与滚动条同步
614 double pos = HorizontalScrollPos;
615 if (-this->GetInternalArrangeOffsetX() > pos) {
616 HorizontalScrollPos = pos;
617 }
618 } else {
619 HorizontalScrollPos = 0;
620 EnableScrollBar(this->Handle, SB_HORZ, ESB_DISABLE_BOTH);
621 _horizontalScrollDisabled = true;
622 }
623 }
624
625 if (VerticalScrollBar) //
626 {
627 double childBottommost = this->GetChildBottommost(true);
628
629 if (int(childBottommost - this->ClientHeight) > 0) {
630 _verticalScrollDisabled = false;
631 EnableScrollBar(this->Handle, SB_VERT, ESB_ENABLE_BOTH);
632 SetVerticalScrollRange(0, childBottommost);
633
634 // 当尺寸改变时确保子元素位置与滚动条同步
635 double pos = VerticalScrollPos;
636 if (-this->GetInternalArrangeOffsetY() > pos) {
637 VerticalScrollPos = pos;
638 }
639 } else {
640 VerticalScrollPos = 0;
641 EnableScrollBar(this->Handle, SB_VERT, ESB_DISABLE_BOTH);
642 _verticalScrollDisabled = true;
643 }
644 }

756 void _MeasureAndArrangeWithoutLayout()
757 {
758 this->GetInternalArrangeOffsetX() = 0;
759 this->GetInternalArrangeOffsetY() = 0;
760
761 int childCount = this->GetChildLayoutCount();
762
763 for (int i = 0; i < childCount; ++i) {
764 // measure
765 UIElement &item = this->GetChildLayoutAt(i);
766 item.Measure(Size{INFINITY, INFINITY});
767 // arrange
768 Size desireSize = item.GetDesireSize();
769 sw::Rect itemRect = item.Rect;
770 Thickness itemMargin = item.Margin;
771 item.Arrange(sw::Rect{itemRect.left - itemMargin.left, itemRect.top - itemMargin.top, desireSize.width, desireSize.height});
772 }
773 }
